20
Token Architecture
How the three layers connect. Change a primitive, the entire system updates.
/* Layer 1: Primitive */
--green-500: oklch(0.76 0.21 150);

/* Layer 2: Semantic — references primitive */
--primary: var(--green-500);
--ring: var(--green-500);

/* Layer 3: Component — references semantic */
--btn-bg: var(--primary);
--btn-hover-bg: var(--primary-hover);
--focus-ring-color: var(--ring);

/* Component class uses component token */
.btn-primary { background: var(--btn-bg); }
.btn-primary:hover { background: var(--btn-hover-bg); }
.btn:focus-visible { box-shadow: 0 0 0 var(--focus-ring-width) var(--focus-ring-color); }
